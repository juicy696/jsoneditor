How to Use in Vue

json-vue.js contains the full editor engine — this single file is all you need.

Direct CDN include (recommended)

<script src="https://unpkg.com/vue@3"></script>
<script src="json-vue.js"></script>

<div id="app">
  <!-- v-model two-way binding: edits sync automatically -->
  <json-editor v-model="config" theme="dark" max_height="50vh" />
</div>
<script>
  Vue.createApp({
    data: () => ({ config: { name: 'Jakey', tags: ['admin'] } })
  }).use(JED.vue)   // one line registers the global <json-editor> component
    .mount('#app');
</script>

Component Props

<json-editor
  v-model="config"       // bound value (object or JSON string)
  title="Title"          // editor title
  theme="dark"           // light | dark, reactive switching supported
  height="420px"         // fixed height (optional)
  max_height="50vh"      // max height, scrolls on overflow, default 50vh
  font_size="13px"       // text size (optional)
  default_view="editor"  // initial view: editor | mindmap | text
  readonly                // read-only mode
/>

npm install

npm install @jed/json-editor

// main.js
import '@jed/json-editor/vue'
