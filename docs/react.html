How to Use in React

json-react.js contains the full editor engine — this single file is all you need.

Direct CDN include (recommended)

<script src="https://unpkg.com/react@18/umd/react.production.min.js"></script>
<script src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>
<script src="json-react.js"></script>

<div id="root"></div>
<script>
  const e = React.createElement;

  function App() {
    const [data, setData] = React.useState({ name: 'Jakey', tags: ['admin'] });
    // unidirectional flow: editor edits flow back through onChange
    return e(JED.react.component, { value: data, onChange: setData, theme: 'dark' });
  }

  ReactDOM.createRoot(document.getElementById('root')).render(e(App));
</script>

Component Props

e(JED.react.component, {
  value: data,            // controlled value (object or JSON string)
  onChange: setData,      // edit callback
  title: 'Title',         // editor title
  theme: 'dark',          // light | dark, rebuilds automatically on change
  height: '420px',        // fixed height (optional)
  max_height: '50vh',     // max height, scrolls on overflow, default 50vh
  font_size: '13px',      // text size (optional)
  default_view: 'editor', // initial view: editor | mindmap | text
  readonly: false         // read-only mode
});
